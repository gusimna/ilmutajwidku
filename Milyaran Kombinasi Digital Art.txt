    <script>
    //<![CDATA[
        // Database komponen prompt global
        const database = {
            styles: [
                "Cyberpunk Aesthetic", "Steampunk Engine Core", "Surrealism oil painting", "Hyper-realistic 3D Digital Art", 
                "Futuristic Solarpunk", "Dark Fantasy Grimdark", "Mythological Ethereal Art", "Synthwave Retro-futurism", 
                "Psychedelic Liquid Ink", "Biomechanical Gigeresque", "Ukiyo-e Modern Fusion", "Baroque Dramatic Chiaroscuro",
                "Minimalist Vector Flat Art", "Gothic Victorian Watercolor", "Vibrant Pop Art Andy Warhol style", "Concept Art Splash Art",
                "Cinematic Moody Film Still", "Abstract Expressionism", "Dieselpunk Blueprint", "Ancient Mythology Fresco",
                "Glitch Art Holographic", "Cosmic Horror Lovecraftian", "Magical Realism", "Stained Glass Windows Style"
            ],
            lighting: [
                "cinematic lighting", "volumetric god rays", "neon glow", "golden hour warmth", "dramatic chiaroscuro",
                "bioluminescent luminescence", "cyberpunk moody neon", "soft diffuse morning light", "rim lighting, epic mood",
                "studio lighting, clean highlights", "mystical twilight shadow", "harsh direct sunlight with deep shadows",
                "quantum particle glow", "iridescent pearlescent light", "ambient occlusion, soft shadows"
            ],
            details: [
                "intricate details", "hyper-detailed textures", "masterpiece composition", "award-winning digital art",
                "breathtaking environment", "highly polished finish", "complex geometry", "octane render crispness",
                "photorealistic fidelity", "epic scale, magnificent", "surreal details", "unreal engine 5 render quality",
                "fine lines and sharp focus", "4k resolution rendering", "impeccable craftsmanship"
            ],
            cameras: [
                "shot on 85mm lens", "macro photography detail", "wide-angle panoramic view", "aerial drone perspective",
                "isometric 3D view", "close-up portrait angle", "dramatic low-angle shot", "orthographic view, architectural render",
                "ray-traced reflections", "depth of field, blurred background", "Hasselblad medium format capture"
            ],
            platforms: [
                "ArtStation trending", "Pixiv masterpiece", "Behance featured", "CGSociety highlight", "DeviantArt top list"
            ]
        };

        const defaultSubjects = ["Cybernetic Deity", "Enchanted Ancient Forest", "Astronaut on Floating Island", "Lost Futuristic City", "Mythical Phoenix"];

        function hitungKombinasi() {
            const total = database.styles.length * database.lighting.length * database.details.length * database.cameras.length * database.platforms.length;
            document.getElementById('math-info').innerText = `Sistem siap mengacak dari sekitar ${total.toLocaleString()}+ kombinasi gaya dasar otomatis (Belum termasuk variasi subjek Anda).`;
        }

        function getRandomElement(arr) {
            return arr[Math.floor(Math.random() * arr.length)];
        }

        // Fungsi utama Generate
        function generate() {
            const subjectInput = document.getElementById('subject').value.trim();
            const finalSubject = subjectInput || getRandomElement(defaultSubjects);
            
            const output = document.getElementById('output');
            output.innerHTML = ""; 
            
            for(let i = 0; i < 5; i++) {
                const style = getRandomElement(database.styles);
                const light = getRandomElement(database.lighting);
                const detail = getRandomElement(database.details);
                const cam = getRandomElement(database.cameras);
                const platform = getRandomElement(database.platforms);
                
                const promptText = `${finalSubject}, ${style}, ${light}, ${detail}, ${cam}, featured on ${platform}, hyperrealistic, sharp focus, --ar 16:9 --v 6.0`;
                
                const card = document.createElement('div');
                card.className = 'prompt-box';
                
                const tag = document.createElement('span');
                tag.className = 'prompt-tag';
                tag.innerText = style.split(' ')[0] + " Art"; 
                
                const textDiv = document.createElement('div');
                textDiv.className = 'prompt-text';
                textDiv.innerText = promptText;
                
                const hint = document.createElement('span');
                hint.className = 'copy-hint';
                hint.innerText = "Klik untuk menyalin prompt 📋";
                
                card.appendChild(tag);
                card.appendChild(textDiv);
                card.appendChild(hint);
                
                card.onclick = () => {
                    navigator.clipboard.writeText(promptText).then(() => {
                        showToast("Prompt berhasil disalin!", false);
                    });
                };
                
                output.appendChild(card);
            }
        }

        // Fitur Baru: Refresh Style (Mengacak ulang gaya dari subjek yang saat ini ada di input)
        function refreshStyles() {
            const subjectInput = document.getElementById('subject').value.trim();
            if (!subjectInput && document.getElementById('output').children.length === 0) {
                showToast("Ketik subjek terlebih dahulu atau langsung klik Generate!", true);
                return;
            }
            generate();
            showToast("Gaya seni berhasil diacak ulang! 🔄", true);
        }

        // Fitur Baru: Reset Form (Membersihkan text input dan daftar prompt)
        function resetForm() {
            document.getElementById('subject').value = "";
            document.getElementById('output').innerHTML = "";
            showToast("Formulir dan hasil dibersihkan! 🗑️", true);
        }

        // Sistem Toast serbaguna
        function showToast(message, isInfo = false) {
            const toast = document.getElementById('toast');
            toast.innerText = message;
            if(isInfo) {
                toast.className = "toast show info";
            } else {
                toast.className = "toast show";
            }
            setTimeout(() => { toast.className = "toast"; }, 2000);
        }

        window.onload = hitungKombinasi;
    //]]>
    </script>
