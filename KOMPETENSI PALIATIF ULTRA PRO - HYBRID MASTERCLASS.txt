                    function toggleFormLock() {
                        const txtArea = document.getElementById('user-thought');
                        const btn = document.getElementById('btn-lock-save');
                        const status = document.getElementById('status-msg');
                        
                        if (txtArea.disabled) {
                            txtArea.disabled = false;
                            btn.className = "btn-action btn-save";
                            btn.innerHTML = "🔒 SIMPAN &amp; KUNCI DATA";
                            status.innerHTML = "✏️ <b>Mode edit aktif.</b> Silakan ubah pokok pikiran/kasus Anda.";
                            txtArea.focus();
                        } else {
                            if(txtArea.value.trim() !== "") {
                                txtArea.disabled = true;
                                btn.className = "btn-action btn-edit";
                                btn.innerHTML = "✏️ EDIT FORMULIR";
