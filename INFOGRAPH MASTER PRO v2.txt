    tagHtml += '<span class="tag" style="background:rgba(0, 255, 163, 0.05); color:#8b949e;">' + s.split(' ')[0] + ' Style</span>';
    tagHtml += '</div>';

    card.innerHTML = tagHtml + '<div class="p-text">' + prompt + '</div><button class="copy" onclick="copyQuantumPrompt(this)">📋 Copy Quantum Prompt</button>';
    fragment.appendChild(card);
  }

  grid.appendChild(fragment);
  status.innerHTML = '🔥 REVOLUTIONARY SUCCESS! Rendered ' + count.toLocaleString('id-ID') + ' hyper-dense quantum infographic engineering prompts.';
}

function copyQuantumPrompt(btn) {
  const text = btn.previousElementSibling.textContent;
  navigator.clipboard.writeText(text).then(() => {
    const orig = btn.textContent;
    btn.textContent = "✓ Prompt Matrix Locked to Clipboard!";
    btn.style.background = "#00ffa3";
    btn.style.color = "#010409";
    setTimeout(() => {
      btn.textContent = orig;
      btn.style.background = "";
      btn.style.color = "";
    }, 2000);
  });
}

}

function downloadPromptDatabase() {
  const texts = document.querySelectorAll('.p-text');
  if (texts.length === 0) return alert("Quantum engine database is currently empty!");
  let content = '⚡ INFOGRAPH MASTER PRO v2.0 — QUANTUM DATABASE EXPORT\nTotal Permutations: ' + texts.length + '\nGeneration Timestamp: ' + new Date().toLocaleString('id-ID') + '\n\n';
  texts.forEach((el, i) => content += '--- QUANTUM DATA MATRIX CONTAINER [' + (i + 1) + '] ---\n' + el.textContent + '\n\n');
  const blob = new Blob([content], {type: "text/plain;charset=utf-8"});
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = 'quantum-infograph-prompts-' + new Date().toISOString().slice(0,10) + '.txt';
  a.click();
}
