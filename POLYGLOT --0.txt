    // Google Client-Side Translation API
    async function translateText(text, fromLang, toLang) {
      if (fromLang === toLang) return text;
      try {
        const url = `https://translate.googleapis.com/translate_a/single?client=gtx&amp;sl=${fromLang}&amp;tl=${toLang}&amp;dt=t&amp;q=${encodeURIComponent(text)}`;
        const response = await fetch(url);
        const data = await response.json();
        return data[0].map(item =&gt; item[0]).join(&#39;&#39;);
      } catch (error) {
        console.error(`Gagal menerjemahkan ke ${toLang}:`, error);
        return `[Gagal menerjemahkan]`;
      }
    }

    // Eksekutor Utama Translasi
    async function executeTranslation() {
      const sourceInput = document.getElementById(&#39;sourceInput&#39;).value.trim();
      const sourceLang = document.getElementById(&#39;sourceLang&#39;).value;
      
      if (!sourceInput) {
        alert(&#39;Mohon masukkan teks prompt terlebih dahulu.&#39;);
        return;
      }

      document.getElementById(&#39;outputID&#39;).value = &#39;Menerjemahkan...&#39;;
      document.getElementById(&#39;outputEN&#39;).value = &#39;Translating...&#39;;
      document.getElementById(&#39;outputAR&#39;).value = &#39;جاري الترجمة...&#39;;
      document.getElementById(&#39;outputJA&#39;).value = &#39;翻訳中...&#39;;

      const [resID, resEN, resAR, resJA] = await Promise.all([
        translateText(sourceInput, sourceLang, &#39;id&#39;),
        translateText(sourceInput, sourceLang, &#39;en&#39;),
        translateText(sourceInput, sourceLang, &#39;ar&#39;),
        translateText(sourceInput, sourceLang, &#39;ja&#39;)
      ]);

      document.getElementById(&#39;outputID&#39;).value = resID;
      document.getElementById(&#39;outputEN&#39;).value = resEN;
      document.getElementById(&#39;outputAR&#39;).value = resAR;
      document.getElementById(&#39;outputJA&#39;).value = resJA;
    }

    // Sistem Salin Clipboard
    function copyField(fieldId, btnId) {
      const targetArea = document.getElementById(fieldId);
      const btn = document.getElementById(btnId);
      
      if (!targetArea.value || targetArea.value.includes(&#39;...&#39;)) return;

      navigator.clipboard.writeText(targetArea.value).then(() =&gt; {
        const originalText = btn.innerHTML;
        btn.innerHTML = `&lt;i data-lucide=&#39;check&#39; class=&#39;w-3.5 h-3.5 text-emerald-400&#39;&gt;&lt;/i&gt; Copied`;
        btn.classList.remove(&#39;text-slate-500&#39;, &#39;bg-slate-900&#39;);
        btn.classList.add(&#39;text-emerald-400&#39;, &#39;bg-emerald-950/60&#39;, &#39;border-emerald-500/30&#39;);
        lucide.createIcons();

        setTimeout(() =&gt; {
          btn.innerHTML = originalText;
          btn.classList.add(&#39;text-slate-500&#39;, &#39;bg-slate-900&#39;);
          btn.classList.remove(&#39;text-emerald-400&#39;, &#39;bg-emerald-950/60&#39;, &#39;border-emerald-500/30&#39;);
          lucide.createIcons();
        }, 2000);
      });
    }
