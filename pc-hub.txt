        function action(type) {
            const care = document.getElementById(&#39;care-selector&#39;).value;
            const status = document.getElementById(&#39;status-msg&#39;);
            let prompt = &quot;&quot;;
            const base = `Bertindaklah sebagai Pakar Perawatan Medis Profesional. Fokus pada ${care}. Berikan analisis yang akurat, empati, dan berbasis bukti klinis.`;
            
            if(type==&quot;emergency&quot;) prompt = `${base} Berikan Emergency Cheat Sheet. Pilih secara acak 1 kondisi kegawatdaruratan yang paling relevan dengan ${care} (bisa berupa krisis fisik atau psikologis), lalu berikan langkah penanganan cepat (poin-poin).`;

            else if(type==&quot;support&quot;) prompt = `${base} Berikan teknik komunikasi empatik untuk mendampingi pasien/keluarga dalam ${care}.`;
            else if(type==&quot;case&quot;) prompt = `${base} Berikan skenario kasus nyata pada ${care} dan minta saya untuk memberikan tindakan terbaik.`;
