    function run() {
      const textInput = document.getElementById('textIn').value.trim() || "Bismillah";
      let count = parseInt(document.getElementById('numIn').value) || 50;
      count = Math.max(1, Math.min(20000, count));

      const grid = document.getElementById('grid');
      grid.innerHTML = '';

      for (let i = 0; i < count; i++) {
        const u = subjects[Math.floor(Math.random()*subjects.length)];
        const k = khats[Math.floor(Math.random()*khats.length)];
        const p = productTypes[Math.floor(Math.random()*productTypes.length)];

    function cp(btn) {
      const text = btn.previousElementSibling.textContent;
      navigator.clipboard.writeText(text).then(() => {
        const orig = btn.textContent;
        btn.textContent = "✓ Tersalin!";
        btn.style.background = "#15803d";
        btn.style.color = "white";
        setTimeout(() => {
          btn.textContent = orig;
          btn.style.background = "";
          btn.style.color = "";
        }, 2000);
      });
    }

    function exportToFile() {
