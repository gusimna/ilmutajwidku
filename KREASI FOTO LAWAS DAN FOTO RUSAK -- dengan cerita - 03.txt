        function renderFilteredPrompts() {
            var selectedEraKey = document.getElementById('eraFilterSelector').value;
            var searchKeyword = document.getElementById('keywordFilterSearch').value.toLowerCase().trim();
            var outputContainer = document.getElementById('jsonOutputArea');
            
            outputContainer.innerHTML = '';
            var eras = Object.keys(timelineData);
            var grandTotalDisplayed = 0;

            for (var e = 0; e < eras.length; e++) {
                var eraKey = eras[e];
                var era = timelineData[eraKey];
                
                if (selectedEraKey !== 'all' && selectedEraKey !== eraKey) continue;

                var itemsInEra = currentGeneratedDatabase.filter(function(item) {
                    var matchEra = item._internalEraKey === eraKey;
                    var matchKeyword = true;
                    if (searchKeyword !== '') {
                        var stringified = JSON.stringify(item).toLowerCase();
                        matchKeyword = stringified.indexOf(searchKeyword) !== -1;
                    }
                    return matchEra && matchKeyword;
                });

                if (itemsInEra.length === 0) continue;
                grandTotalDisplayed += itemsInEra.length;

                var eraHeader = document.createElement('div');
                eraHeader.className = 'pt-4 pb-1 border-b border-slate-700 text-indigo-400 font-tech font-bold text-[11px] sticky top-0 bg-slate-950 z-10 flex justify-between items-center';
                eraHeader.innerHTML = '<span>DIGITAL ERA TIMELINE: ' + era.name.toUpperCase() + '</span><span class="text-slate-500 text-[9px] font-mono">' + itemsInEra.length + ' PLUGINS</span>';
                outputContainer.appendChild(eraHeader);

                for (var i = 0; i < itemsInEra.length; i++) {
                    var item = itemsInEra[i];
                    var card = document.createElement('div');
                    card.className = 'bg-slate-900 border border-slate-800 rounded-xl p-4 flex flex-col gap-3 shadow-md hover:border-slate-700 transition-all';
                    
                    var visualCleanObj = {
                        id: item.id,
                        digital_era: item.era,
                        date: item.date,
                        pose_style: item.pose,
                        demographic_age: item.age,
                        social_vibe: item.vibe,
                        mood_tone: item.humor,
                        lens_rotation: item.frame_rotation_degree,
                        ai_parameters: item.ai_parameters, 
                        blueprint_visual: item.narrative_content,
                        prompt: item.prompt
                    };

                    var jsonStr = JSON.stringify(visualCleanObj, null, 4);
                    var rawSagaCommand = buildSagaStoryCommand(item);
                    
                    card.innerHTML = '\
                        <div class="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-2 border-b border-slate-800 pb-2">\
                            <span class="text-[10px] font-mono font-bold text-indigo-400">' + item.id + '</span>\
                            <div class="flex gap-2 w-full sm:w-auto justify-end">\
                                <button class="bg-indigo-600 hover:bg-indigo-700 text-white px-2.5 py-1 rounded-md text-[10px] font-mono font-medium transition-all cursor-pointer border border-indigo-500 shadow-xs btn-copy-json" data-json="' + btoa(unescape(encodeURIComponent(jsonStr))) + '">Tombol 1: Copy JSON Blueprint</button>\
                                <button class="bg-slate-800 hover:bg-stone-700 text-zinc-300 hover:text-white px-2.5 py-1 rounded-md text-[10px] font-mono font-medium transition-all cursor-pointer border border-slate-700 shadow-xs btn-copy-saga" data-saga="' + btoa(unescape(encodeURIComponent(rawSagaCommand))) + '">Tombol 2: Copy Perintah Cerita</button>\
                            </div>\
                        </div>\
                        <div class="bg-slate-950 p-3 rounded-lg border border-slate-850 max-h-48 overflow-y-auto custom-scrollbar">\
                            <pre class="font-mono text-emerald-400 whitespace-pre-wrap text-[9px]">' + jsonStr + '</pre>\
                        </div>\
                    ';
                    outputContainer.appendChild(card);
                }
            }

            if (grandTotalDisplayed === 0) {
                outputContainer.innerHTML = '<div class="text-slate-600 font-mono text-center py-8">// Tidak ditemukan prompt JSON modern yang cocok dengan filter.</div>';
            } else {
                var jsonButtons = outputContainer.getElementsByClassName('btn-copy-json');
                for (var j = 0; j < jsonButtons.length; j++) {
                    jsonButtons[j].onclick = function() {
                        var decodedJson = decodeURIComponent(escape(atob(this.getAttribute('data-json'))));

                        var self = this; var oldTxt = self.innerText; self.innerText = "Saga Command Copied! ✓"; self.style.backgroundColor = "#9d174d";
                        setTimeout(function() { self.innerText = oldTxt; self.style.backgroundColor = "#1e293b"; }, 1500);
                    };
                }
            }
        }

        function generateContemporaryPrompts() {
            var subject = document.getElementById('subjectInput').value.trim();
            if(!subject) {
                alert("Subjek Utama tidak boleh kosong untuk melakukan kompilasi data!");
                return;
            }
            var attachedImageUrl = document.getElementById('imageUrlInput').value.trim();
            var imgSimilarityWeight = document.getElementById('imageSimilarityInput').value;
            var poseValue = document.getElementById('poseStyleInput').value;
            var ageValue = document.getElementById('ageStyleInput').value;
            var vibeValue = document.getElementById('vibeStyleInput').value;
            var humorValue = document.getElementById('humorStyleInput').value;
            var tiltValue = document.getElementById('tiltStyleInput').value;
            var chosenCondition = document.getElementById('conditionInput').value;
            var conditionText = filterModifiers[chosenCondition];
            currentGeneratedDatabase = [];

            var calculatedImageWeight = (parseInt(imgSimilarityWeight) / 50).toFixed(2);
            var currentPoseObj = getPoseModifier(poseValue);
            var currentAgeObj = getAgeModifier(ageValue);
            var currentVibeObj = getVibeModifier(vibeValue);
            var currentHumorObj = getHumorModifier(humorValue);
            var currentTiltObj = getTiltModifier(tiltValue);
            
            var promptCounter = 1;
            var eras = Object.keys(timelineData);

            for (var e = 0; e < eras.length; e++) {
                var eraKey = eras[e];
                var era = timelineData[eraKey];

                for (var i = 1; i <= 26; i++) {
                    var styleVariant = "";
                    var activeFilter = conditionText;
                    
                    var mod = i % 5;
                    var structuralBase = "Atmospheric setting: " + currentPoseObj.phrase + ". Cast properties: " + currentAgeObj.phrase + ". Cultural context: " + currentVibeObj.phrase + ". Mood preset: " + currentHumorObj.phrase + ". Spatial rotation alignment: " + currentTiltObj.phrase + ".";
                    if (mod === 0) styleVariant = "Cinematic candid raw shot of " + subject + ". " + structuralBase;
                    else if (mod === 1) styleVariant = "Ultra crisp urban documentation snapshot of " + subject + ". " + structuralBase;
                    else if (mod === 2) styleVariant = "Wide-angle panoramic establishing frame of " + subject + ". " + structuralBase;
                    else if (mod === 3) styleVariant = "Neon noir low-light ambiance exposure study of " + subject + ". " + structuralBase;
                    else if (mod === 4) styleVariant = "Minimalist social media feed aesthetic capture of " + subject + ". " + structuralBase;
                    
                    var fakePhotoDate = generateFakeDate(era.start_year, era.end_year);
                    var fullPrompt = (attachedImageUrl !== "" ? attachedImageUrl + " " : "") + "A high-fidelity contemporary photograph dated " + fakePhotoDate + ", visualizing " + styleVariant + " Rendered via " + era.tech + ". Production specs: " + era.camera_spec + ", stylistic filter overlay: " + activeFilter + ". --ar 16:9 --v 6.0" + (attachedImageUrl !== "" ? " --iw " + calculatedImageWeight : "");
                    
                    var promptId = "CYBER-PRMPT-" + String(promptCounter).padStart(3, '0');

                    var aiParametersMetadata = {
                        target_model: "Midjourney v6.0 / Stable Diffusion XL Ultra",
                        aspect_ratio: "16:9 (Widescreen)",
                        cfg_scale: 6.5,
                        negative_prompt: "old vintage sepia, black and white monochrome, physical photo scratches, damaged paper, 3d blender render, low resolution artifacting, historical clothing",
                        sampler_type: "DPM++ SDE Karras",
                        seed_randomization: Math.floor(Math.random() * 4294967295),
                        image_weight: attachedImageUrl !== "" ? parseFloat(calculatedImageWeight) : null
                    };

                    currentGeneratedDatabase.push({
                        id: promptId,
                        era: era.name,
                        date: fakePhotoDate, 
                        pose: currentPoseObj.label,
                        age: currentAgeObj.label, 
                        vibe: currentVibeObj.label,
                        humor: currentHumorObj.label,
                        frame_rotation_degree: currentTiltObj.label,
                        ai_parameters: aiParametersMetadata, 
                        narrative_content: buildNarrativeContent(subject, fakePhotoDate, era.name, currentVibeObj.concept, currentAgeObj.label, currentHumorObj.mood, parseInt(tiltValue), currentTiltObj.type),
                        prompt: fullPrompt,
                        _internalEraKey: eraKey
                    });
                    promptCounter++;
                }
            }
            
            document.getElementById('filterBarContainer').classList.remove('hidden');
            document.getElementById('btnBulkCopyJSON').classList.remove('hidden');
            document.getElementById('eraFilterSelector').value = 'all';
            document.getElementById('keywordFilterSearch').value = '';

            renderFilteredPrompts();
        }
